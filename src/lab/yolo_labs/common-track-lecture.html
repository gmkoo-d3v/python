03 · 합성 영상

사람을 오려서 걷게 만들기

Penn-Fudan 검증 사진의 사람을 마스크로 오려 RGBA 스프라이트를 만들고, 배경 위를 걷게 합니다. 누가 누구인지 우리가 아니까 정답 ID를 붙일 수 있어요.

common_track.py · 42–98행핵심 발췌
def _sprites() -> list[np.ndarray]:
    """검증 사진의 사람을 마스크로 오려 RGBA 조각으로 만든다. 사진 가장자리에 잘린 사람은 뺀다."""
    sprites = []
    for image in sorted((YOLO_ROOT / "valid" / "images").glob("*.png")):
        rgb = np.array(Image.open(PENN / "PNGImages" / image.name).convert("RGB"))
        mask = np.array(Image.open(PENN / "PedMasks" / f"{image.stem}_mask.png"))
        for k in range(1, int(mask.max()) + 1):
            ys, xs = np.nonzero(mask == k)
            if len(ys) == 0:
                continue
            y1, y2, x1, x2 = ys.min(), ys.max() + 1, xs.min(), xs.max() + 1
            touches = x1 == 0 or y1 == 0 or x2 == mask.shape[1] or y2 == mask.shape[0]
            if touches or y2 - y1 < 150:
                continue
            alpha = (mask[y1:y2, x1:x2] == k).astype(np.uint8) * 255
            sprites.append(np.dstack([rgb[y1:y2, x1:x2], alpha]))
    return sprites


def _plan_walkers(rng: np.random.Generator, count: int) -> list[dict]:
    """사람마다 걷는 방향, 빠르기, 발 높이(가까울수록 아래·크게), 나타나는 시각을 정한다."""
    walkers = []
    for i in range(count):
        foot = int(rng.choice([250, 275, 300, 330, 355]))     # 같은 줄을 마주 걷는 사람끼리 겹친다
        height = int(110 + (foot - 250) * 1.0)
        walkers.append({"id": i + 1, "dir": 1 if i % 2 == 0 else -1, "speed": float(rng.uniform(5, 11)),
                        "foot": foot, "height": height, "start": int(rng.integers(0, FRAMES - 70)),
                        "phase": float(rng.uniform(0, 6.28))})
    return walkers

스프라이트: 검증 사진 34장에서 사람별 RGBA 조각. 마스크의 사람 픽셀만 alpha=255.

가장자리·작은 사람 제외: 사진 밖으로 잘린 사람(touches)과 높이 150px 미만은 뺍니다.

걷는 계획: 발 높이 5줄 중 하나, 키는 발 높이에 비례(110~215px).

방향·속도: 짝수 ID는 오른쪽, 홀수는 왼쪽. 속도 5~11 px/장면.

시작·위상: 0~90장면 사이에 등장, 걸음 오르내림 위상은 무작위.

그리기: 먼 사람부터(발 높이 오름차순) 그려 앞 사람이 뒤 사람을 덮게. 기둥은 마지막에.

def make_video() -> tuple[list[np.ndarray], list[list[dict]]]:
    """합성 영상의 장면들과 장면마다의 정답(ID, 보이는 부분의 상자, 보이는 비율)을 만든다."""
    sprites = _sprites()
    rng = np.random.default_rng(SEED)
    order = rng.permutation(len(sprites))[:PEOPLE]
    walkers = _plan_walkers(rng, PEOPLE)
    background = _background()
    frames, truth = [], []
    for t in range(FRAMES):
        frame = background.copy()
        owner = np.zeros((HEIGHT, WIDTH), np.int16)           # 픽셀마다 맨 앞에 보이는 사람의 ID
        full = {}
        for walker in sorted(walkers, key=lambda w: w["foot"]):    # 먼 사람부터 그린다
            placed = _place(walker, sprites[order[walker["id"] - 1]], t)
            if placed is None:
                continue
            x, y, piece = placed
            fx1, fy1 = max(x, 0), max(y, 0)
            fx2, fy2 = min(x + piece.shape[1], WIDTH), min(y + piece.shape[0], HEIGHT)
            if fx2 <= fx1 or fy2 <= fy1:
                continue
            crop = piece[fy1 - y:fy2 - y, fx1 - x:fx2 - x]
            alpha = crop[..., 3:4] / 255.0
            region = frame[fy1:fy2, fx1:fx2]
            frame[fy1:fy2, fx1:fx2] = (alpha * crop[..., :3] + (1 - alpha) * region).astype(np.uint8)
            owner[fy1:fy2, fx1:fx2][crop[..., 3] > 127] = walker["id"]
            full[walker["id"]] = int((crop[..., 3] > 127).sum())
        frame[:, PILLAR[0]:PILLAR[1]] = (70, 72, 78)          # 기둥은 사람 앞에 있다
        owner[:, PILLAR[0]:PILLAR[1]] = 0
        boxes = []
        for pid, total in full.items():
            ys, xs = np.nonzero(owner == pid)
            if len(ys) == 0 or len(ys) / total < MIN_VISIBLE or ys.max() - ys.min() < MIN_TRUTH_HEIGHT:
                continue
            boxes.append({"id": pid, "box": [int(xs.min()), int(ys.min()), int(xs.max()) + 1, int(ys.max()) + 1],
                          "visible": round(len(ys) / total, 3)})
        frames.append(frame)
        truth.append(boxes)
    return frames, truth
🔬 정확히는 — alpha 합성과 owner

frame = alpha * 사람 + (1-alpha) * 배경으로 가장자리를 부드럽게 합칩니다. owner는 “이 픽셀을 마지막으로 덮은 사람 ID”입니다. 기둥을 그린 뒤 owner[:, 392:432]=0으로 지우면, 기둥 뒤 픽셀은 정답에서 사라집니다. 그래서 visible이 줄어요.

⚠️ 함정
  • int(mask.max())+1 범위로 사람 ID를 순회. 마스크에 빈 ID가 있으면 건너뜁니다.
  • _place는 아직 등장 전이거나 화면을 완전히 지나갔으면 None. start가 늦으면 영상에 못 나올 수도.
  • 기둥은 모든 사람 앞에 그려집니다. 기둥 뒤 사람은 잠깐 완전히 사라져 검출도 안 됩니다.
  • 정답 박스는 “보이는 픽셀” 기준. 가려진 사람의 박스는 실제 몸보다 작습니다.
  • 배경·사람 배치는 seed 11로 고정. make_video를 다시 불러도 같은 영상.

def synth() -> tuple[list[np.ndarray], list[list[dict]]]:
    """합성 영상과 정답. 씨앗이 정해져 있어 매번 같은 장면이 나온다.

    mp4는 손실 압축이라 다시 읽으면 원본과 조금 다르므로, 검출은 늘 다시 그린 원본 장면에 한다.

    frames, truth = make_video()
    if not VIDEO.exists():
        save_video(frames, VIDEO)

def detect_frames(frames: list[np.ndarray], conf: float = DET_CONF) -> list[np.ndarray]:
    """장면마다 2일차 best.pt로 사람을 찾는다. 결과는 (N, 5) = x1, y1, x2, y2, 점수."""
    require_lab5()
    from ultralytics import YOLO
    model = YOLO(str(BEST))
    device = pick_device()
    results = []
    for frame in frames:
        r = model.predict(frame, imgsz=IMAGE_SIZE, conf=conf, device=device, verbose=False)[0]
        boxes = r.boxes.xyxy.cpu().numpy()
        scores = r.boxes.conf.cpu().numpy()[:, None]
        results.append(np.hstack([boxes, scores]).astype(np.float32))
    return results


def synth_detections() -> list[np.ndarray]:
    """합성 영상의 검출 결과. 한 번 돌려 runs/track/dets.npz에 두고 다시 쓴다."""
    if DETS.exists():
        saved = np.load(DETS)
        return [saved[f"f{t}"] for t in range(FRAMES)]
    frames, _ = synth()
    dets = detect_frames(frames)
    np.savez(DETS, **{f"f{t}": d for t, d in enumerate(dets)})
    return dets

05 · iou_matrix() · match()

겹침 표와 최고의 짝

모든 “이전 상자 × 이번 상자”의 IoU를 한 번에 계산하고, 합이 가장 큰 1:1 짝을 헝가리안 방법으로 찾습니다.

def iou_matrix(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """(N,4)와 (M,4) xyxy 상자의 IoU 표(1일차 lab1과 같은 계산)."""
    if len(a) == 0 or len(b) == 0:
        return np.zeros((len(a), len(b)))
    x1 = np.maximum(a[:, None, 0], b[None, :, 0])
    y1 = np.maximum(a[:, None, 1], b[None, :, 1])
    x2 = np.minimum(a[:, None, 2], b[None, :, 2])
    y2 = np.minimum(a[:, None, 3], b[None, :, 3])
    inter = np.clip(x2 - x1, 0, None) * np.clip(y2 - y1, 0, None)
    area_a = (a[:, 2] - a[:, 0]) * (a[:, 3] - a[:, 1])
    area_b = (b[:, 2] - b[:, 0]) * (b[:, 3] - b[:, 1])
    return inter / (area_a[:, None] + area_b[None, :] - inter)


def match(iou: np.ndarray, min_iou: float) -> tuple[list[tuple[int, int]], list[int], list[int]]:
    """IoU 합이 가장 큰 짝을 헝가리안 방법으로 찾고, min_iou 아래인 짝은 버린다."""
    if iou.size == 0:
        return [], list(range(iou.shape[0])), list(range(iou.shape[1]))
    rows, cols = linear_sum_assignment(-iou)
    pairs = [(r, c) for r, c in zip(rows, cols) if iou[r, c] >= min_iou]
    left_rows = sorted(set(range(iou.shape[0])) - {r for r, _ in pairs})
    left_cols = sorted(set(range(iou.shape[1])) - {c for _, c in pairs})
    return pairs, left_rows, left_cols

🔬 정확히는 — 헝가리안

linear_sum_assignment(-iou)는 “행마다 하나, 열마다 하나” 짝 중 합이 최대가 되는 조합을 찾습니다. -iou를 넣는 이유는 이 함수가 최소화 문제를 풀기 때문. 짝을 지은 뒤 min_iou 미만은 버리고, 남은 행/열을 “짝 없는 트랙/검출”로 돌려줍니다. 탐욕법은 큰 IoU부터 집어서 전체 합이 최적이 아닐 수 있습니다.

  • min_iou 필터는 할당 후에 합니다. 먼저 버리면 최적 조합이 달라질 수 있어요.

  • 박스가 (0,0,0,0)이면 area 0 → 0으로 나눌 수 있습니다. 실제 검출은 보통 양수 크기.

def evaluate(truth: list[list[dict]], tracks: list[list[tuple[int, list[float]]]], thr: float = MATCH_IOU) -> dict:
    """CLEAR MOT(MOTA, ID 바뀜)과 IDF1을 계산한다.

    MOTA: 장면마다 정답과 트랙을 짝짓는다. 앞 장면의 짝이 아직 겹치면 그대로 두고, 나머지를 헝가리안으로.
          정답의 짝이 앞에서 짝이던 트랙과 다르면 ID 바뀜 1번. MOTA = 1 - (놓침 + 헛짚음 + ID 바뀜) / 정답 수.
    IDF1: 영상 전체에서 정답 한 사람과 트랙 하나를 1:1로 묶어(겹친 장면 수가 가장 많게), 그 짝이 맞힌 장면 비율.
    """
    last, fn, fp, switches, total = {}, 0, 0, 0, 0
    overlap: dict[tuple[int, int], int] = {}
    gt_count: dict[int, int] = {}
    trk_count: dict[int, int] = {}
    for gt, trk in zip(truth, tracks):
        g_ids = [g["id"] for g in gt]
        g_boxes = np.array([g["box"] for g in gt], float).reshape(-1, 4)
        t_ids = [i for i, _ in trk]
        t_boxes = np.array([b for _, b in trk], float).reshape(-1, 4)
        iou = iou_matrix(g_boxes, t_boxes)
        total += len(gt)
        for gi in g_ids:
            gt_count[gi] = gt_count.get(gi, 0) + 1
        for ti in t_ids:
            trk_count[ti] = trk_count.get(ti, 0) + 1
        for r, c in zip(*np.nonzero(iou >= thr)):
            overlap[(g_ids[r], t_ids[c])] = overlap.get((g_ids[r], t_ids[c]), 0) + 1
        pairs = []
        free_g, free_t = set(range(len(gt))), set(range(len(trk)))
        for r, gi in enumerate(g_ids):                      # 앞 장면의 짝을 먼저 이어 간다
            if gi in last and last[gi] in t_ids:
                c = t_ids.index(last[gi])
                if c in free_t and iou[r, c] >= thr:
                    pairs.append((r, c))
                    free_g.discard(r)
                    free_t.discard(c)
        rows, cols = sorted(free_g), sorted(free_t)
        sub_pairs, _, _ = match(iou[np.ix_(rows, cols)], thr)
        for r, c in sub_pairs:
            gi, ti = g_ids[rows[r]], t_ids[cols[c]]
            if gi in last and last[gi] != ti:
                switches += 1
            pairs.append((rows[r], cols[c]))
        for r, c in pairs:
            last[g_ids[r]] = t_ids[c]
        fn += len(gt) - len(pairs)
        fp += len(trk) - len(pairs)
    g_list, t_list = sorted(gt_count), sorted(trk_count)
    score = np.zeros((len(g_list), len(t_list)))
    for (gi, ti), n in overlap.items():
        score[g_list.index(gi), t_list.index(ti)] = n
    rows, cols = linear_sum_assignment(-score) if score.size else ([], [])
    idtp = float(score[rows, cols].sum()) if score.size else 0.0
    idf1 = 2 * idtp / (sum(gt_count.values()) + sum(trk_count.values())) if total else 0.0
    return {"MOTA": 1 - (fn + fp + switches) / total, "IDF1": idf1, "ID바뀜": switches,
            "놓침": fn, "헛짚음": fp, "트랙수": len(trk_count), "정답": total}

장면별 매칭: 정답 박스와 트랙 박스의 IoU 표. thr=0.5 이상만 후보.

연속성 우선: 정답 gi가 앞 장면에서 트랙 last[gi]와 짝이었다면, 이번에도 그 트랙이 겹칠 때 먼저 잇습니다.

나머지는 헝가리안: 자유로운 정답/트랙끼리 match.

ID 바뀜: 정답이 이전에 다른 트랙과 짝이었는데 이번에 새 트랙으로 바뀌면 +1.

놓침/헛짚음: 짝이 없는 정답 수, 짝이 없는 트랙 수를 누적.

MOTA: 1 - (FN+FP+IDSW)/정답수. 1에 가까울수록 좋고, 음수도 가능.

IDF1: 영상 전체에서 정답-트랙 쌍의 겹친 장면 수를 점수로 두고, 헝가리안으로 1:1 묶음. 2*IDTP/(정답수+트랙수).

🔬 정확히는 — MOTA vs IDF1

MOTA는 장면 단위의 검출·연결 실수를 모두 벌점으로 줍니다. IDF1은 영상 전체에서 “정답 한 사람이 트랙 하나에 얼마나 오래 붙어 있었나”를 봅니다. ID가 자주 바뀌면 MOTA도 떨어지지만 IDF1이 더 민감하게 반응해요. 두 숫자를 같이 봐야 합니다.

⚠️ 함정
  • last[gi]는 정답 ID별 “마지막으로 짝지은 트랙”. 영상 전체에서 유지됩니다.
  • ID 바뀜은 “이전과 다른 트랙”일 때만 셉니다. 같은 트랙이 잠깐 끊겼다 돌아오면? 새 트랙 ID면 바뀜.
  • total=0이면 MOTA 계산에서 0으로 나눌 수 있습니다. 실제로는 정답이 있습니다.
  • IDF1의 score는 “겹친 장면 수”이지 IoU 합이 아닙니다. 매칭의 질보다 지속성을 봅니다.
  • show()는 PEOPLE을 “실제 사람 수”로 출력. 트랙 수와 비교해 과분할을 봅니다.

class IoUTracker:
    def __init__(self, conf: float = CONF, min_iou: float = MIN_IOU):
        self.conf, self.min_iou = conf, min_iou
        self.tracks: dict[int, np.ndarray] = {}     # ID → 마지막 상자
        self.next_id = 1

    def update(self, dets: np.ndarray) -> list[tuple[int, list[float]]]:
        boxes = dets[dets[:, 4] >= self.conf, :4]
        ids = list(self.tracks)
        old = np.array([self.tracks[i] for i in ids]).reshape(-1, 4)
        pairs, _, new = match(iou_matrix(old, boxes), self.min_iou)
        self.tracks = {ids[r]: boxes[c] for r, c in pairs}          # 짝이 없는 트랙은 사라진다
        for c in new:
            self.tracks[self.next_id] = boxes[c]
            self.next_id += 1
        return [(i, b.tolist()) for i, b in self.tracks.items()]

class KalmanTrack:
    """상태 8칸: 가운데·크기 4칸과 그 빠르기 4칸. 한 장면에 빠르기만큼 움직인다고 본다(등속)."""

    F = np.eye(8) + np.eye(8, k=4)                 # 다음 위치 = 지금 위치 + 빠르기
    H = np.eye(4, 8)                               # 재는 것은 위치·크기 4칸뿐
    Q = np.diag([1, 1, 1, 1, 0.5, 0.5, 0.1, 0.1]) ** 2     # 움직임이 등속에서 벗어나는 정도
    R = np.diag([4, 4, 8, 8]) ** 2                         # 검출 상자의 흔들림(픽셀)

    def __init__(self, box: np.ndarray, track_id: int):
        self.x = np.r_[to_state(box), np.zeros(4)]
        self.P = np.diag([10, 10, 10, 10, 100, 100, 100, 100]) ** 2.0
        self.id, self.hits, self.missed = track_id, 1, 0

    def predict(self) -> np.ndarray:
        self.x = self.F @ self.x
        self.P = self.F @ self.P @ self.F.T + self.Q
        self.missed += 1
        return to_box(self.x)

    def correct(self, box: np.ndarray) -> None:
        y = to_state(box) - self.H @ self.x
        S = self.H @ self.P @ self.H.T + self.R
        K = self.P @ self.H.T @ np.linalg.inv(S)
        self.x = self.x + K @ y
        self.P = (np.eye(8) - K @ self.H) @ self.P
        self.hits += 1
        self.missed = 0

상태 8칸: 중심 x, y, 너비, 높이 + 각각의 속도 4칸.

F: 다음 상태 = 지금 + 속도. 등속 운동 가정.

H: 실제로 측정하는 건 위치·크기 4칸뿐.

Q: 등속에서 벗어나는 잡음. 위치 1, 속도 0.5, 크기 속도 0.1.

R: 검출 박스의 흔들림. x/y 4px, w/h 8px.

predict: 상태와 공분산을 한 장면 전진, missed += 1.

correct: 검출과의 차이(y)로 칼만 이득 K를 계산해 상태 수정. missed=0, hits+=1.

class SortTracker:
    def __init__(self, conf: float = CONF, min_iou: float = MIN_IOU, max_age: int = MAX_AGE, min_hits: int = MIN_HITS):
        self.conf, self.min_iou, self.max_age, self.min_hits = conf, min_iou, max_age, min_hits
        self.tracks: list[KalmanTrack] = []
        self.next_id = 1

    def new_track(self, box: np.ndarray) -> None:
        self.tracks.append(KalmanTrack(box, self.next_id))
        self.next_id += 1

    def output(self) -> list[tuple[int, list[float]]]:
        """이번 장면에 검출로 확인했고, 충분히 여러 번 본 트랙만 내보낸다."""
        return [(t.id, t.box().tolist()) for t in self.tracks if t.missed == 0 and t.hits >= self.min_hits]

    def update(self, dets: np.ndarray) -> list[tuple[int, list[float]]]:
        boxes = dets[dets[:, 4] >= self.conf, :4]
        predicted = np.array([t.predict() for t in self.tracks]).reshape(-1, 4)
        pairs, _, new = match(iou_matrix(predicted, boxes), self.min_iou)
        for r, c in pairs:
            self.tracks[r].correct(boxes[c])
        for c in new:
            self.new_track(boxes[c])
        self.tracks = [t for t in self.tracks if t.missed <= self.max_age]
        return self.output()

⚠️ 함정
  • predict()가 missed를 먼저 올립니다. 그래서 짝이 맞으면 correct()가 0으로 되돌립니다.
  • 새 트랙은 첫 장면에 hits=1이라 min_hits=2면 다음 장면부터 출력됩니다.
  • max_age=0이면 한 장면만 놓쳐도 삭제 → IoU 추적과 비슷해짐.
  • 칼만 필터는 등속 가정. 갑자기 방향을 바꾸면 예측이 틀립니다.
  • 트랙 인덱스 r은 tracks 리스트 순서. 삭제 후 다음 장면에서 인덱스가 바뀝니다.

낮은 점수 검출을 한 번 더 쓰기 (ByteTrack)

반쯤 가려진 사람은 점수가 낮게 나옵니다. ByteTrack은 1차로 높은 점수와 모든 트랙을 잇고, 2차로 남은 트랙과 낮은 점수를 잇습니다.

HIGH = 0.5
LOW = 0.1
NEW = 0.6
FIRST_IOU = 0.2       # 1차 짝짓기(원 논문의 match_thresh 0.8 = 1 - IoU)
SECOND_IOU = 0.5      # 2차 짝짓기는 더 엄격하게(낮은 검출에는 헛검출이 섞여 있다)
MAX_AGE = 30          # 원 논문의 track_buffer(30장면)


class ByteTracker(SortTracker):
    def __init__(self, high: float = HIGH, low: float = LOW, new: float = NEW, max_age: int = MAX_AGE):
        super().__init__(conf=high, max_age=max_age, min_hits=1)
        self.high, self.low, self.new = high, low, new

    def update(self, dets: np.ndarray) -> list[tuple[int, list[float]]]:
        high = dets[dets[:, 4] >= self.high]
        low = dets[(dets[:, 4] >= self.low) & (dets[:, 4] < self.high)]
        predicted = np.array([t.predict() for t in self.tracks]).reshape(-1, 4)
        pairs, left_tracks, left_high = match(iou_matrix(predicted, high[:, :4]), FIRST_IOU)
        for r, c in pairs:
            self.tracks[r].correct(high[c, :4])
        # 2차: 바로 앞 장면까지 보던 트랙만(오래 잃은 트랙에 낮은 검출을 붙이면 엉뚱한 사람과 이어진다)
        recent = [r for r in left_tracks if self.tracks[r].missed == 1]
        pairs2, _, _ = match(iou_matrix(predicted[recent], low[:, :4]), SECOND_IOU)
        for r, c in pairs2:
            self.tracks[recent[r]].correct(low[c, :4])
        for c in left_high:
            if high[c, 4] >= self.new:
                self.new_track(high[c, :4])
        self.tracks = [t for t in self.tracks if t.missed <= self.max_age]
        return self.output()

두 더미: 고득점(≥0.5)과 저득점(0.1~0.5).

1차 매칭: 예측 vs 고득점, IoU 0.2. 맞은 트랙은 correct.

2차 후보: 1차에서 남았고 missed==1인 트랙만. 바로 앞 장면까지 보던 트랙.

2차 매칭: 그 트랙들 vs 저득점, IoU 0.5로 더 엄격.

새 트랙: 1차에서 남은 고득점 중 ≥0.6만. 저득점으로는 새 트랙을 만들지 않음.

정리: missed > 30 트랙 삭제. 출력은 min_hits=1이라 바로.

🔬 정확히는 — 왜 2차가 더 엄격?

저득점 검출에는 헛검출(배경, 그림자)이 섞여 있습니다. 그래서 IoU 0.5로 더 확실할 때만 잇습니다. 또 “오래 잃은 트랙”에는 저득점을 붙이지 않습니다(missed==1만). 엉뚱한 사람과 이어지는 것을 막는 장치예요.

⚠️ 함정
  • ByteTrack은 SORT를 상속하지만 min_hits=1로 바꿉니다. 첫 장면부터 출력.
  • 1차에서 맞은 트랙도 predicted는 이미 predict()로 전진한 상태. correct가 위치를 보정.
  • left_high는 1차에서 남은 고득점의 인덱스. high[c,4]로 점수 확인.
  • low=HIGH로 두면 저득점 더미가 비어 2차 매칭이 사라집니다. lab15의 ablation.
  • 원 논문의 match_thresh=0.8은 “1-IoU” 기준. 여기선 IoU 0.2로 단순화.

def count_crossings(frames_of_items: list[list[tuple[int, list[float]]]], line_x: float) -> tuple[int, int]:
    """ID마다 마지막으로 본 가운데 x를 기억해 두고, 선의 반대편에서 다시 보이면 한 번 넘은 것으로 센다."""
    last_x: dict[int, float] = {}
    right, left = 0, 0
    for items in frames_of_items:
        for track_id, box in items:
            cx = (box[0] + box[2]) / 2
            if track_id in last_x:
                if last_x[track_id] < line_x <= cx:
                    right += 1
                elif cx < line_x <= last_x[track_id]:
                    left += 1
            last_x[track_id] = cx
    return right, left

def recognition(frames: list[np.ndarray], truth: list[list[dict]], embed, shaded: bool = False) -> tuple[float, int]:
    inside = lambda g: g["box"][0] > 2 and g["box"][2] < WIDTH - 2      # 화면 가장자리에 잘린 모습은 뺀다
    gallery, gallery_id, queries, query_id = [], [], [], []
    for pid in range(1, PEOPLE + 1):
        seen = [(t, g) for t, boxes in enumerate(truth) for g in boxes if g["id"] == pid and inside(g)]
        clear = [(t, g) for t, g in seen if g["visible"] >= 0.8] or seen   # 늘 가려진 사람은 보이는 대로
        if not clear:
            continue
        first = clear[:GALLERY_SHOTS]
        later = [(t, g) for t, g in seen if t >= first[-1][0] + LATER and g["visible"] >= 0.5]  # 없으면 질문도 없다
        for t, g in first:
            gallery.append(crop(frames[t], g["box"]))
            gallery_id.append(pid)
        for t, g in later[::max(1, len(later) // QUERIES)][:QUERIES]:
            view = crop(frames[t], g["box"])
            queries.append(shade(view) if shaded else view)
            query_id.append(pid)
    gal, qry = embed(gallery), embed(queries)
    guess = np.array(gallery_id)[np.argmax(qry @ gal.T, axis=1)]
    return float((guess == np.array(query_id)).mean()), len(queries)

def iou_matrix(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """(N,4)와 (M,4) xyxy 상자의 IoU 표(1일차 lab1과 같은 계산)."""
    if len(a) == 0 or len(b) == 0:
        return np.zeros((len(a), len(b)))
    x1 = np.maximum(a[:, None, 0], b[None, :, 0])
    y1 = np.maximum(a[:, None, 1], b[None, :, 1])
    x2 = np.minimum(a[:, None, 2], b[None, :, 2])
    y2 = np.minimum(a[:, None, 3], b[None, :, 3])
    inter = np.clip(x2 - x1, 0, None) * np.clip(y2 - y1, 0, None)
    area_a = (a[:, 2] - a[:, 0]) * (a[:, 3] - a[:, 1])
    area_b = (b[:, 2] - b[:, 0]) * (b[:, 3] - b[:, 1])
    return inter / (area_a[:, None] + area_b[None, :] - inter)


def match(iou: np.ndarray, min_iou: float) -> tuple[list[tuple[int, int]], list[int], list[int]]:
    """IoU 합이 가장 큰 짝을 헝가리안 방법으로 찾고, min_iou 아래인 짝은 버린다."""
    if iou.size == 0:
        return [], list(range(iou.shape[0])), list(range(iou.shape[1]))
    rows, cols = linear_sum_assignment(-iou)
    pairs = [(r, c) for r, c in zip(rows, cols) if iou[r, c] >= min_iou]
    left_rows = sorted(set(range(iou.shape[0])) - {r for r, _ in pairs})
    left_cols = sorted(set(range(iou.shape[1])) - {c for _, c in pairs})
    return pairs, left_rows, left_cols

    return [tracker.update(d) for d in dets]

def evaluate(truth: list[list[dict]], tracks: list[list[tuple[int, list[float]]]], thr: float = MATCH_IOU) -> dict:
    """CLEAR MOT(MOTA, ID 바뀜)과 IDF1을 계산한다.

    MOTA: 장면마다 정답과 트랙을 짝짓는다. 앞 장면의 짝이 아직 겹치면 그대로 두고, 나머지를 헝가리안으로.
          정답의 짝이 앞에서 짝이던 트랙과 다르면 ID 바뀜 1번. MOTA = 1 - (놓침 + 헛짚음 + ID 바뀜) / 정답 수.
    IDF1: 영상 전체에서 정답 한 사람과 트랙 하나를 1:1로 묶어(겹친 장면 수가 가장 많게), 그 짝이 맞힌 장면 비율.
    """
    last, fn, fp, switches, total = {}, 0, 0, 0, 0
    overlap: dict[tuple[int, int], int] = {}
    gt_count: dict[int, int] = {}
    trk_count: dict[int, int] = {}
    for gt, trk in zip(truth, tracks):
        g_ids = [g["id"] for g in gt]
        g_boxes = np.array([g["box"] for g in gt], float).reshape(-1, 4)
        t_ids = [i for i, _ in trk]
        t_boxes = np.array([b for _, b in trk], float).reshape(-1, 4)
        iou = iou_matrix(g_boxes, t_boxes)
        total += len(gt)
        for gi in g_ids:
            gt_count[gi] = gt_count.get(gi, 0) + 1
        for ti in t_ids:
            trk_count[ti] = trk_count.get(ti, 0) + 1
        for r, c in zip(*np.nonzero(iou >= thr)):
            overlap[(g_ids[r], t_ids[c])] = overlap.get((g_ids[r], t_ids[c]), 0) + 1
        pairs = []
        free_g, free_t = set(range(len(gt))), set(range(len(trk)))
        for r, gi in enumerate(g_ids):                      # 앞 장면의 짝을 먼저 이어 간다
            if gi in last and last[gi] in t_ids:
                c = t_ids.index(last[gi])
                if c in free_t and iou[r, c] >= thr:
                    pairs.append((r, c))
                    free_g.discard(r)
                    free_t.discard(c)
        rows, cols = sorted(free_g), sorted(free_t)
        sub_pairs, _, _ = match(iou[np.ix_(rows, cols)], thr)
        for r, c in sub_pairs:
            gi, ti = g_ids[rows[r]], t_ids[cols[c]]
            if gi in last and last[gi] != ti:
                switches += 1
            pairs.append((rows[r], cols[c]))
        for r, c in pairs:
            last[g_ids[r]] = t_ids[c]
        fn += len(gt) - len(pairs)
        fp += len(trk) - len(pairs)
    g_list, t_list = sorted(gt_count), sorted(trk_count)
    score = np.zeros((len(g_list), len(t_list)))
    for (gi, ti), n in overlap.items():
        score[g_list.index(gi), t_list.index(ti)] = n
    rows, cols = linear_sum_assignment(-score) if score.size else ([], [])
    idtp = float(score[rows, cols].sum()) if score.size else 0.0
    idf1 = 2 * idtp / (sum(gt_count.values()) + sum(trk_count.values())) if total else 0.0
    return {"MOTA": 1 - (fn + fp + switches) / total, "IDF1": idf1, "ID바뀜": switches,
            "놓침": fn, "헛짚음": fp, "트랙수": len(trk_count), "정답": total}


def show(name: str, m: dict) -> None:

          f"놓침 {m['놓침']:4d}  헛짚음 {m['헛짚음']:4d}  만든 ID {m['트랙수']:3d}개(실제 {PEOPLE}명)")

class IoUTracker:
    def __init__(self, conf: float = CONF, min_iou: float = MIN_IOU):

그래서 잠깐 놓치거나 기둥 뒤로 가려져도 같은 ID로 다시 잡을 수 있다.
"""
import numpy as np

CONF = 0.5
MIN_IOU = 0.3
MAX_AGE = 15      # 이만큼(1.5초) 못 찾으면 트랙을 버린다
MIN_HITS = 2      # 이만큼 잇달아 찾아야 내보낸다(한 번 튄 헛검출은 ID를 받지 않게)


def to_state(box: np.ndarray) -> np.ndarray:
    """xyxy 상자 → (가운데 x, 가운데 y, 너비, 높이)."""
    return np.array([(box[0] + box[2]) / 2, (box[1] + box[3]) / 2, box[2] - box[0], box[3] - box[1]])


def to_box(state: np.ndarray) -> np.ndarray:
    cx, cy, w, h = state[:4]
    return np.array([cx - w / 2, cy - h / 2, cx + w / 2, cy + h / 2])


class KalmanTrack:
    """상태 8칸: 가운데·크기 4칸과 그 빠르기 4칸. 한 장면에 빠르기만큼 움직인다고 본다(등속)."""

    F = np.eye(8) + np.eye(8, k=4)                 # 다음 위치 = 지금 위치 + 빠르기
    H = np.eye(4, 8)                               # 재는 것은 위치·크기 4칸뿐
    Q = np.diag([1, 1, 1, 1, 0.5, 0.5, 0.1, 0.1]) ** 2     # 움직임이 등속에서 벗어나는 정도
    R = np.diag([4, 4, 8, 8]) ** 2                         # 검출 상자의 흔들림(픽셀)

    def __init__(self, box: np.ndarray, track_id: int):
        self.x = np.r_[to_state(box), np.zeros(4)]
        self.P = np.diag([10, 10, 10, 10, 100, 100, 100, 100]) ** 2.0
        self.id, self.hits, self.missed = track_id, 1, 0

    def predict(self) -> np.ndarray:
        self.x = self.F @ self.x
        self.P = self.F @ self.P @ self.F.T + self.Q
        self.missed += 1
        return to_box(self.x)

    def correct(self, box: np.ndarray) -> None:
        y = to_state(box) - self.H @ self.x
        S = self.H @ self.P @ self.H.T + self.R
        K = self.P @ self.H.T @ np.linalg.inv(S)
        self.x = self.x + K @ y
        self.P = (np.eye(8) - K @ self.H) @ self.P
        self.hits += 1
        self.missed = 0

    def box(self) -> np.ndarray:
        return to_box(self.x)


class SortTracker:
    def __init__(self, conf: float = CONF, min_iou: float = MIN_IOU, max_age: int = MAX_AGE, min_hits: int = MIN_HITS):
        self.conf, self.min_iou, self.max_age, self.min_hits = conf, min_iou, max_age, min_hits
        self.tracks: list[KalmanTrack] = []
        self.next_id = 1

    def new_track(self, box: np.ndarray) -> None:
        self.tracks.append(KalmanTrack(box, self.next_id))
        self.next_id += 1

    def output(self) -> list[tuple[int, list[float]]]:
        """이번 장면에 검출로 확인했고, 충분히 여러 번 본 트랙만 내보낸다."""
        return [(t.id, t.box().tolist()) for t in self.tracks if t.missed == 0 and t.hits >= self.min_hits]

    def update(self, dets: np.ndarray) -> list[tuple[int, list[float]]]:
        boxes = dets[dets[:, 4] >= self.conf, :4]
        predicted = np.array([t.predict() for t in self.tracks]).reshape(-1, 4)
        pairs, _, new = match(iou_matrix(predicted, boxes), self.min_iou)
        for r, c in pairs:
            self.tracks[r].correct(boxes[c])
        for c in new:
            self.new_track(boxes[c])
        self.tracks = [t for t in self.tracks if t.missed <= self.max_age]
        return self.output()


def main() -> None:
    from lab13_iou_track import IoUTracker
    _, truth = synth()

    main()

"""14차시: ByteTrack, 그리고 세 추적기 견주기.

SORT는 점수가 낮은 검출을 버린다. 그런데 반쯤 가려진 사람은 점수가 낮게 나온다.
ByteTrack은 짝짓기를 두 번 한다.
  1차: 점수 높은 검출(≥ HIGH)과 모든 트랙
  2차: 1차에서 짝을 못 찾은 트랙과 점수 낮은 검출(LOW ~ HIGH)
새 트랙은 점수가 아주 높은 검출(≥ NEW)에서만 만든다. 낮은 검출은 '이미 따라가던 사람'을 잇는 데만 쓴다.
"""
import numpy as np

from common_track import evaluate, iou_matrix, match, run_tracker, show, synth, synth_detections
from lab13_iou_track import IoUTracker
from lab14_sort import SortTracker

HIGH = 0.5
LOW = 0.1
NEW = 0.6
FIRST_IOU = 0.2       # 1차 짝짓기(원 논문의 match_thresh 0.8 = 1 - IoU)
SECOND_IOU = 0.5      # 2차 짝짓기는 더 엄격하게(낮은 검출에는 헛검출이 섞여 있다)
MAX_AGE = 30          # 원 논문의 track_buffer(30장면)


class ByteTracker(SortTracker):
    def __init__(self, high: float = HIGH, low: float = LOW, new: float = NEW, max_age: int = MAX_AGE):
        super().__init__(conf=high, max_age=max_age, min_hits=1)
        self.high, self.low, self.new = high, low, new

    def update(self, dets: np.ndarray) -> list[tuple[int, list[float]]]:
        high = dets[dets[:, 4] >= self.high]
        low = dets[(dets[:, 4] >= self.low) & (dets[:, 4] < self.high)]
        predicted = np.array([t.predict() for t in self.tracks]).reshape(-1, 4)
        pairs, left_tracks, left_high = match(iou_matrix(predicted, high[:, :4]), FIRST_IOU)
        for r, c in pairs:
            self.tracks[r].correct(high[c, :4])
        # 2차: 바로 앞 장면까지 보던 트랙만(오래 잃은 트랙에 낮은 검출을 붙이면 엉뚱한 사람과 이어진다)
        recent = [r for r in left_tracks if self.tracks[r].missed == 1]
        pairs2, _, _ = match(iou_matrix(predicted[recent], low[:, :4]), SECOND_IOU)
        for r, c in pairs2:
            self.tracks[recent[r]].correct(low[c, :4])
        for c in left_high:
            if high[c, 4] >= self.new:
                self.new_track(high[c, :4])
        self.tracks = [t for t in self.tracks if t.missed <= self.max_age]
        return self.output()


def main() -> None:
    _, truth = synth()
    dets = synth_detections()
    results = {
        "IoU 추적": evaluate(truth, run_tracker(IoUTracker(), dets)),
        "SORT": evaluate(truth, run_tracker(SortTracker(), dets)),
        "ByteTrack": evaluate(truth, run_tracker(ByteTracker(), dets)),
    }
    for name, m in results.items():
        show(name, m)
    print()
    print("2차 짝짓기를 끄면(낮은 검출을 쓰지 않으면)")
    show("  LOW=HIGH", evaluate(truth, run_tracker(ByteTracker(low=HIGH), dets)))


if __name__ == "__main__":
    main()

COLORS = [(230, 25, 75), (60, 180, 75), (255, 225, 25), (0, 130, 200), (245, 130, 48), (145, 30, 180),
          (70, 240, 240), (240, 50, 230), (210, 245, 60), (250, 190, 212), (0, 128, 128), (170, 110, 40)]


def count_crossings(frames_of_items: list[list[tuple[int, list[float]]]], line_x: float) -> tuple[int, int]:
    """ID마다 마지막으로 본 가운데 x를 기억해 두고, 선의 반대편에서 다시 보이면 한 번 넘은 것으로 센다."""
    last_x: dict[int, float] = {}
    right, left = 0, 0
    for items in frames_of_items:
        for track_id, box in items:
            cx = (box[0] + box[2]) / 2
            if track_id in last_x:
                if last_x[track_id] < line_x <= cx:
                    right += 1
                elif cx < line_x <= last_x[track_id]:
                    left += 1
            last_x[track_id] = cx
    return right, left


def draw(frames: list[np.ndarray], tracks: list[list[tuple[int, list[float]]]], line_x: int) -> list[np.ndarray]:
    """상자와 ID, 세는 선, 지금까지 넘은 수를 그린다."""
    drawn, seen = [], []
    for frame, items in zip(frames, tracks):
        seen.append(items)
        right, left = count_crossings(seen, line_x)
        canvas = frame.copy()
        cv2.line(canvas, (line_x, 0), (line_x, canvas.shape[0]), (255, 255, 255), 2)
        for track_id, box in items:
            x1, y1, x2, y2 = (int(v) for v in box)
            color = COLORS[track_id % len(COLORS)]
            cv2.rectangle(canvas, (x1, y1), (x2, y2), color, 2)
            cv2.putText(canvas, str(track_id), (x1, max(y1 - 4, 12)), cv2.FONT_HERSHEY_SIMPLEX, 0.6, color, 2)
        cv2.putText(canvas, f"-> {right}   <- {left}", (10, 24), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (255, 255, 255), 2)
        drawn.append(canvas)
    return drawn

"""19차시: 겉모습으로 같은 사람 찾기(수요일 추적 영상에 DINOv3 더하기).

3일차 추적기는 위치와 움직임만 봤다. 그래서 기둥 뒤로 오래 가렸다 나온 사람은 새 ID를 받았다.
1) 알아보기: 사람마다 처음 몇 장면의 모습을 '사진첩'에 넣고, 나중 장면의 모습이 사진첩의 누구와 가장
   닮았는지 맞혀 본다. DINOv3 특징과 색 히스토그램(HSV)을 견준다. 나중 모습을 그늘에 들어간 것처럼
   어둡고 푸르게 바꿔서도 해 본다(조명이 바뀌는 실제 현장).
2) 이어 붙이기: 추적기가 끊어 놓은 트랙 가운데, 앞 트랙이 끝나고 GAP 장면 안에, 걷던 빠르기로 갔을
   자리 근처(REACH 픽셀)에서 시작한 트랙이 겉모습도 THRESHOLD보다 닮았으면 같은 ID로 잇는다.
lab5(best.pt)와 3일차 공통 코드(common_track.py, lab14·15)를 씁니다.
"""
import cv2
import numpy as np
from PIL import Image

from common_dino import load
from common_dino import embed as dino_embed
from common_track import PEOPLE, WIDTH, evaluate, run_tracker, show, synth, synth_detections
from lab14_sort import SortTracker
from lab15_bytetrack import ByteTracker

GALLERY_SHOTS = 3        # 사진첩에 넣는 처음 모습 수(사람마다)
LATER = 30               # 사진첩 마지막 장면에서 이만큼 지난 뒤의 모습을 질문으로
QUERIES = 10             # 사람마다 질문 수
GAP = 50                 # 앞 트랙이 끝나고 이 장면 수 안에 시작한 트랙만 이을 후보
REACH = 60               # 걷던 대로 갔을 자리에서 이만큼(픽셀) 안에서 시작해야 후보
THRESHOLD = 0.8          # 트랙 평균 특징의 코사인 유사도가 이보다 크면 잇는다
CROPS_PER_TRACK = 10
PERSON_SIZE = (256, 128) # 사람 조각은 세로로 길게(정사각으로 늘이지 않는다)
SHADE = (0.55, 0.6, 0.75)    # 그늘: R·G·B에 곱하는 값(어둡고 푸르게)


def crop(frame: np.ndarray, box: list[float]) -> Image.Image:
    x1, y1, x2, y2 = (int(round(v)) for v in box)
    h, w = frame.shape[:2]
    x1, y1, x2, y2 = max(x1, 0), max(y1, 0), min(x2, w), min(y2, h)
    return Image.fromarray(frame[y1:max(y2, y1 + 2), x1:max(x2, x1 + 2)])


def hsv_hist(images: list[Image.Image]) -> np.ndarray:
    """견줄 거리: 색상·채도·명도 8×8×8칸 히스토그램(길이 1로 맞춤)."""
    out = []
    for im in images:
        hsv = cv2.cvtColor(np.asarray(im), cv2.COLOR_RGB2HSV)
        h = cv2.calcHist([hsv], [0, 1, 2], None, [8, 8, 8], [0, 180, 0, 256, 0, 256]).ravel()
        out.append(np.sqrt(h / max(h.sum(), 1)))
    return np.array(out)


def shade(image: Image.Image) -> Image.Image:
    a = np.asarray(image, np.float32) * np.array(SHADE)
    return Image.fromarray(np.clip(a, 0, 255).astype(np.uint8))


def recognition(frames: list[np.ndarray], truth: list[list[dict]], embed, shaded: bool = False) -> tuple[float, int]:
    inside = lambda g: g["box"][0] > 2 and g["box"][2] < WIDTH - 2      # 화면 가장자리에 잘린 모습은 뺀다
    gallery, gallery_id, queries, query_id = [], [], [], []
    for pid in range(1, PEOPLE + 1):
        seen = [(t, g) for t, boxes in enumerate(truth) for g in boxes if g["id"] == pid and inside(g)]
        clear = [(t, g) for t, g in seen if g["visible"] >= 0.8] or seen   # 늘 가려진 사람은 보이는 대로
        if not clear:
            continue
        first = clear[:GALLERY_SHOTS]
        later = [(t, g) for t, g in seen if t >= first[-1][0] + LATER and g["visible"] >= 0.5]  # 없으면 질문도 없다
        for t, g in first:
            gallery.append(crop(frames[t], g["box"]))
            gallery_id.append(pid)
        for t, g in later[::max(1, len(later) // QUERIES)][:QUERIES]:
            view = crop(frames[t], g["box"])
            queries.append(shade(view) if shaded else view)
            query_id.append(pid)
    gal, qry = embed(gallery), embed(queries)
    guess = np.array(gallery_id)[np.argmax(qry @ gal.T, axis=1)]
    return float((guess == np.array(query_id)).mean()), len(queries)


def stitch(frames: list[np.ndarray], tracks: list[list[tuple[int, list[float]]]], embed) -> list[list[tuple[int, list[float]]]]:
    """끊긴 트랙을 겉모습으로 잇는다. 시작이 이른 트랙부터, 가장 닮은 '이미 끝난' 트랙에 붙인다."""
    spans: dict[int, list[tuple[int, list[float]]]] = {}
    for t, items in enumerate(tracks):
        for tid, box in items:
            spans.setdefault(tid, []).append((t, box))
    feature = {}
    for tid, seen in spans.items():
        picks = seen[::max(1, len(seen) // CROPS_PER_TRACK)][:CROPS_PER_TRACK]
        f = embed([crop(frames[t], b) for t, b in picks]).mean(axis=0)
        feature[tid] = f / np.linalg.norm(f)
    start = {tid: s[0][0] for tid, s in spans.items()}
    end = {tid: s[-1][0] for tid, s in spans.items()}
    cx = lambda box: (box[0] + box[2]) / 2

    def reachable(old: int, new: int) -> bool:
        """앞 트랙이 마지막 빠르기로 계속 걸었다면 새 트랙이 시작한 자리 근처(REACH 픽셀 안)에 와 있어야 한다."""
        seen = spans[old]
        back = seen[-min(len(seen), 5)]
        speed = (cx(seen[-1][1]) - cx(back[1])) / max(seen[-1][0] - back[0], 1)
        guess = cx(seen[-1][1]) + speed * (start[new] - end[old])
        return abs(cx(spans[new][0][1]) - guess) < REACH

    root, taken = {tid: tid for tid in spans}, set()
    for tid in sorted(spans, key=start.get):
        candidates = [p for p in spans if p not in taken and end[p] < start[tid] <= end[p] + GAP and reachable(p, tid)]
        if not candidates:
            continue
        best = max(candidates, key=lambda p: feature[p] @ feature[tid])
        if feature[best] @ feature[tid] > THRESHOLD:
            root[tid] = root[best]
            taken.add(best)
    return [[(root[tid], box) for tid, box in items] for items in tracks]


def main() -> None:
    frames, truth = synth()
    dets = synth_detections()
    model = load()
    dino = lambda ims: dino_embed(model, ims, size=PERSON_SIZE)
    print("1) 알아보기: 처음 모습(사람마다 3장)으로 만든 사진첩에서 나중 모습의 주인 맞히기")
    print(f"   {'':<8}{'그대로':>8}{'그늘에서':>10}   (아무렇게나 찍으면 {1 / PEOPLE:.3f})")
    for name, fn in [("DINOv3", dino), ("색", hsv_hist)]:
        same, n = recognition(frames, truth, fn)
        dark, _ = recognition(frames, truth, fn, shaded=True)
        print(f"   {name:<8}{same:>8.3f}{dark:>10.3f}   (질문 {n}장)")
    print(f"\n2) 이어 붙이기(앞 트랙이 끝나고 {GAP}장면 안, 걷던 자리에서 {REACH}픽셀 안, 코사인 유사도 > {THRESHOLD})")
    for name, tracker in [("SORT", SortTracker()), ("ByteTrack", ByteTracker())]:
        tracks = run_tracker(tracker, dets)
        show(name, evaluate(truth, tracks))
        show("  +DINOv3", evaluate(truth, stitch(frames, tracks, dino)))
        show("  +색", evaluate(truth, stitch(frames, tracks, hsv_hist)))


if __name__ == "__main__":
    main()

15 · 용어 사전

        
막히면 여기로

      

      
MOTA
CLEAR MOT 지표. 1 - (놓침 + 헛짚음 + ID 바뀜) / 정답 수. 1에 가까울수록 좋고 음수도 가능.

      
IDF1
영상 전체에서 정답과 트랙을 1:1로 묶었을 때, 그 짝이 맞힌 장면 비율. 2·IDTP/(정답수+트랙수).

      
ID 바뀜 (ID switch)
같은 정답 사람이 이전 장면과 다른 트랙 ID로 짝지어지는 것. 추적기가 이름표를 바꾼 횟수.

      
헝가리안 매칭
행/열마다 하나씩 짝지어 합을 최대(또는 최소)로 만드는 조합. scipy.optimize.linear_sum_assignment.

      
칼만 필터
예측(predict)과 보정(correct)을 반복하며 상태를 추정하는 필터. SORT는 위치·크기와 그 속도를 8칸 상태로 둡니다.

      
SORT
Simple Online and Realtime Tracking. 칼만 예측 + IoU 매칭. max_age로 잠깐 기다리고 min_hits로 헛검출을 거릅니다.

      
ByteTrack
고득점 1차 매칭 + 저득점 2차 매칭. 낮은 점수의 가려진 사람을 이어 붙이는 데 강합니다.

      
트랙 (track)
같은 사람이라고 이어 붙인 검출의 시간적 묶음. 고유 ID를 가집니다.

      
스프라이트 (sprite)
마스크로 오려 낸 사람 RGBA 조각. 합성 영상에 붙입니다.

      
owner mask
픽셀마다 “맨 앞에 보이는 사람 ID”를 적은 배열. 기둥에 가린 부분은 0으로 지웁니다.

      
선 넘기 (line crossing)
트랙 중심 x가 세로선을 어느 방향으로 통과했는지 세는 것. ID별 마지막 x만 기억하면 됩니다.

      
ReID
Re-identification. 겉모습 특징으로 같은 사람을 다시 찾는 것. lab20은 DINOv3와 HSV 히스토그램을 비교합니다.

      
gallery / query
gallery는 “아는 사람 사진첩”, query는 “누구인지 물어볼 사진”. 특징 내적으로 가장 가까운 gallery를 고릅니다.

      
코사인 유사도
두 벡터의 방향이 얼마나 같은지. embed가 길이를 1로 맞추면 내적과 같습니다. DINO 특징 비교에 사용.
